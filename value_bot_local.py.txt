def send_telegram_message(message):
    url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
    payload = {
        "chat_id": CHAT_ID,
        "text": message,
        "parse_mode": "Markdown"
    }
    try:
        response = requests.post(url, json=payload, timeout=10)
        if response.status_code == 200:
            print("✅ Alert wysłany na Telegram pomyślnie!")
        else:
            print(f"❌ Błąd wysyłania: {response.text}")
    except Exception as e:
        print(f"❌ Wyjątek przy wysyłaniu na Telegram: {e}")

def check_value_bets():
    print("🔍 Skanuję rynki dla wielu dyscyplin sportowych...")
    
    for sport in SPORTS_LIST:
        print(f"➡️ Sprawdzam sport: {sport}")
        url = f"https://api.the-odds-api.com/v4/sports/{sport}/odds/?apiKey={ODDS_API_KEY}&regions={REGION}&markets=h2h"
        
        try:
            response = requests.get(url, timeout=10)
            if response.status_code != 200:
                print(f"Błąd dla {sport}: {response.status_code} - {response.text}")
                continue  # Przechodzi do kolejnego sportu w przypadku błędu

            matches = response.json()
            print(f"Pobrano {len(matches)} meczów dla {sport}.")

            for match in matches:
                home_team = match['home_team']
                away_team = match['away_team']
                
                bookmakers = match.get('bookmakers', [])
                if not bookmakers:
                    continue
                    
                bookie = bookmakers[0]
                bookie_name = bookie['title']
                market = bookie['markets'][0]
                outcomes = {out['name']: out['price'] for out in market['outcomes']}
                
                home_odds = outcomes.get(home_team)
                
                # Przykładowy warunek value (np. kurs > 1.40)
                if home_odds and home_odds > 1.40:
                    message = (
                        f"🔥 **MULTI-SPORT VALUE ALERT** 🔥\n\n"
                        f"🏆 **Sport/Liga:** `{sport}`\n"
                        f"⚽ **Mecz:** {home_team} vs {away_team}\n"
                        f"🎯 **Typ:** {home_team} (Gospodarz)\n"
                        f"📊 **Kurs ({bookie_name}):** `{home_odds}`\n"
                        f"💰 **Stawka:** 50 PLN (Flat)\n\n"
                        f"🤖 *Bot 24/7 na Railway (Cykl co 2h).*"
                    )
                    
                    print(f"Znaleziono value: {home_team} vs {away_team} ({sport}) - Kurs: {home_odds}")
                    send_telegram_message(message)
                        
        except Exception as e:
            print(f"❌ Wyjątek przy {sport}: {e}")
