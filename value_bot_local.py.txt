def load_data():
    if not os.path.exists(DATA_FILE):
        return {"bets": [], "units_balance": 0.0, "pln_balance": 0.0}
    try:
        with open(DATA_FILE, "r") as f:
            return json.load(f)
    except Exception:
        return {"bets": [], "units_balance": 0.0, "pln_balance": 0.0}

def save_data(data):
    with open(DATA_FILE, "w") as f:
        json.dump(data, f, indent=4)

def send_telegram_message(message):
    url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage"
    payload = {
        "chat_id": CHAT_ID,
        "text": message,
        "parse_mode": "Markdown"
    }
    try:
        response = requests.post(url, json=payload, timeout=10)
        if response.status_code == 200:
            res_json = response.json()
            return res_json.get("result", {}).get("message_id")
    except Exception as e:
        print(f"❌ Błąd Telegrama: {e}")
    return None

def edit_telegram_message(message_id, new_message):
    if not message_id:
        return
    url = f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/editMessageText"
    payload = {
        "chat_id": CHAT_ID,
        "message_id": message_id,
        "text": new_message,
        "parse_mode": "Markdown"
    }
    try:

    except Exception as e:
        print(f"❌ Błąd edycji wiadomości: {e}")

def check_value_bets(data):
    print("🔍 Skanuję rynki pod kątem Betclica...")
    
    for sport in SPORTS_LIST:
        url = f"https://api.the-odds-api.com/v4/sports/{sport}/odds/?apiKey={ODDS_API_KEY}&regions={REGION}&markets=h2h"
        try:
            response = requests.get(url, timeout=10)
            if response.status_code != 200:
                continue

            matches = response.json()
            for match in matches:
                match_id = match['id']
                
                if any(b['match_id'] == match_id for b in data['bets']):
                    continue

                home_team = match['home_team']
                away_team = match['away_team']
                
                bookmakers = match.get('bookmakers', [])
                if not bookmakers:
                    continue
                
                home_odds_list = []
                betclic_odds = None
                
                for bookie in bookmakers:
                    bookie_name = bookie['title']
                    markets = bookie.get('markets', [])
                    for m in markets:
                        if m['key'] == 'h2h':
                            outcomes = {out['name']: out['price'] for out in m['outcomes']}
                            h_odd = outcomes.get(home_team)
                            if h_odd:
                                home_odds_list.append(h_odd)
                                if "betclic" in bookie_name.lower():
                                    betclic_odds = h_odd

                if not home_odds_list or not betclic_odds:
                    continue

                avg_market_odds = sum(home_odds_list) / len(home_odds_list)

                if betclic_odds >= 1.50 and betclic_odds >= avg_market_odds:
                    initial_message = (
                        f"⚡ **PROFIT_BOT // VIP PICK** ⚡\n"
                        f"━━━━━━━━━━━━━━━━━━━━━━\n"
                        f"📌 **Dyscyplina:** `{sport.upper()}`\n"
                        f"⚔️ **Spotkanie:** `{home_team} vs {away_team}`\n"
                        f"🎯 **Typ:** `{home_team} (Gospodarz)`\n"
                        f"📊 **Kurs (Betclic):** `{betclic_odds}`\n"
                        f"💵 **Stawka:** `25 PLN` (1j)\n"
                        f"━━━━━━━━━━━━━━━━━━━━━━\n"
                        f"⏳ **STATUS: Oczekiwanie na rozstrzygnięcie...**\n\n"
                        f"📈 **Bilans serii:** `{data['units_balance']:+.2f}j` | `CZIKATA: {data['pln_balance']:+.2f} PLN`\n"
                        f"🤖 *Autonomous Value Engine v2.4*"
                    )
                    
                    msg_id = send_telegram_message(initial_message)
                    if msg_id:
                        print(f"🎯 Znaleziono bet na Betclic: {home_team} vs {away_team} po kursie {betclic_odds}")
                        data['bets'].append({
                            "match_id": match_id,
                            "sport": sport,
                            "home_team": home_team,
                            "away_team": away_team,
                            "odds": betclic_odds,
                            "telegram_message_id": msg_id
                        })
                        save_data(data)
                        break 
                
        except Exception as e:
            print(f"❌ Wyjątek przy skanowaniu Betclica dla {sport}: {e}")
